import asyncio
import json
import re
from parsers.browser_proxy_client import BrowserProxyClient

async def main():
    client = BrowserProxyClient()
    html = await client.get_html(
        'https://www.ozon.ru/product/elektronnaya-kniga-7-amazon-kindle-bez-reklamy-2024-gen11-paperwhite-16gb-garantiya-12-mesyatsev-4406016672/'
    )
    
    print('=== ВСЕ ЦЕНЫ В ТЕКСТЕ ===')
    prices = re.findall(r'(\d[\d\s]+)\s*₽', html)
    for p in prices[:30]:
        print(' ', p)
    
    print('\n=== JSON-LD ===')
    for m in re.findall(r'<script[^>]*type="application/ld\+json"[^>]*>(.*?)</script>', html, re.DOTALL):
        try:
            d = json.loads(m)
            if d.get('@type') == 'Product':
                print('  offers:', d.get('offers'))
        except:
            pass
    
    print('\n=== СЛОВА РЯДОМ С 17905, 19894, 36000 ===')
    for line in html.split('\n'):
        if '₽' in line and any(x in line for x in ['17905', '19894', '36000', 'банк', 'стало', 'С банками']):
            print(line[:400])
    
    await client.close()

asyncio.run(main())
