import asyncio
import random
import httpx
from typing import List, Optional, Set
from datetime import datetime, timedelta


class ProxyRotator:
    """Загрузка, валидация и ротация бесплатных HTTP-прокси."""

    SOURCES = [
        "https://api.proxyscrape.com/v2/?request=get&protocol=http&timeout=10000&country=all&ssl=all&anonymity=all",
        "https://www.proxy-list.download/api/v1/get?type=http",
        "https://raw.githubusercontent.com/TheSpeedX/PROXY-List/master/http.txt",
        "https://raw.githubusercontent.com/monosans/proxy-list/main/proxies/http.txt",
    ]

    def __init__(self, enabled: bool = True, max_proxies: int = 20):
        self.enabled = enabled
        self.max_proxies = max_proxies
        self.proxies: List[str] = []
        self.working_proxies: List[str] = []
        self.current_index = 0
        self.last_fetch: Optional[datetime] = None

    async def fetch_proxies(self) -> List[str]:
        all_proxies: Set[str] = set()
        async with httpx.AsyncClient(timeout=15.0) as client:
            for source in self.SOURCES:
                try:
                    resp = await client.get(source)
                    if resp.status_code == 200:
                        for line in resp.text.splitlines():
                            line = line.strip()
                            if line and ":" in line and not line.startswith("#"):
                                parts = line.split(":")
                                if len(parts) >= 2:
                                    ip, port = parts[0], parts[1]
                                    if ip.replace(".", "").isdigit():
                                        all_proxies.add(f"http://{ip}:{port}")
                except Exception as e:
                    print(f"[ProxyRotator] Ошибка загрузки {source}: {e}")

        self.proxies = list(all_proxies)[: self.max_proxies * 3]
        self.last_fetch = datetime.now()
        print(f"[ProxyRotator] Загружено {len(self.proxies)} прокси")
        return self.proxies

    async def validate_proxy(self, proxy: str) -> bool:
        try:
            async with httpx.AsyncClient(proxies=proxy, timeout=8.0) as client:
                resp = await client.get("http://httpbin.org/ip")
                return resp.status_code == 200
        except Exception:
            return False

    async def validate_all(self, max_concurrent: int = 10):
        if not self.proxies:
            await self.fetch_proxies()

        self.working_proxies = []
        semaphore = asyncio.Semaphore(max_concurrent)

        async def check_one(proxy: str):
            async with semaphore:
                if await self.validate_proxy(proxy):
                    self.working_proxies.append(proxy)
                    print(f"[ProxyRotator] ✅ {proxy}")
                else:
                    print(f"[ProxyRotator] ❌ {proxy}")

        await asyncio.gather(*[check_one(p) for p in self.proxies])
        self.proxies = []
        print(f"[ProxyRotator] Рабочих: {len(self.working_proxies)}")

    def get_next(self) -> Optional[str]:
        if not self.working_proxies:
            return None
        proxy = self.working_proxies[self.current_index % len(self.working_proxies)]
        self.current_index += 1
        return proxy

    async def get_working_proxy(self) -> Optional[str]:
        if not self.enabled:
            return None
        if not self.working_proxies or (
            self.last_fetch and datetime.now() - self.last_fetch > timedelta(minutes=30)
        ):
            await self.fetch_proxies()
            await self.validate_all()
        return self.get_next()

    def remove_current(self, proxy: str):
        if proxy in self.working_proxies:
            self.working_proxies.remove(proxy)
            print(f"[ProxyRotator] Удалён забаненный: {proxy}")

import httpx
import asyncio
import json
import random
from typing import Optional, Dict, Any
from .stealth_config import USER_AGENTS, STEALTH_SCRIPT


class BrowserWingClient:
    def __init__(self, base_url: str = "http://127.0.0.1:8080/api/v1", proxy: Optional[str] = None):
        self.base_url = base_url
        self.proxy = proxy
        self.client = httpx.AsyncClient(base_url=base_url, timeout=60.0, proxies=proxy)
        self.user_agent = random.choice(USER_AGENTS)

    async def navigate(self, url: str, wait: float = 5.0) -> bool:
        try:
            response = await self.client.post("/executor/navigate", json={"url": url})
            response.raise_for_status()
            await asyncio.sleep(wait)
            await self.apply_stealth()
            return True
        except Exception as e:
            print(f"[BrowserWing] Ошибка navigate: {e}")
            return False

    async def apply_stealth(self):
        try:
            await self.execute_js(STEALTH_SCRIPT)
        except Exception as e:
            print(f"[BrowserWing] Stealth ошибка: {e}")

    async def execute_js(self, script: str) -> Any:
        try:
            response = await self.client.post("/executor/execute_script", json={"script": script})
            response.raise_for_status()
            data = response.json()
            return data.get("result") if isinstance(data, dict) else data
        except Exception as e:
            print(f"[BrowserWing] Ошибка execute_js: {e}")
            return None

    async def snapshot(self) -> Dict[str, Any]:
        try:
            response = await self.client.get("/executor/snapshot")
            response.raise_for_status()
            return response.json()
        except Exception as e:
            print(f"[BrowserWing] Ошибка snapshot: {e}")
            return {}

    async def get_html(self) -> str:
        try:
            result = await self.execute_js("return document.documentElement.outerHTML")
            return str(result) if result else ""
        except Exception:
            return ""

    async def set_browser_proxy(self, proxy: str) -> bool:
        """Пытается установить прокси в BrowserWing (если API поддерживает)."""
        try:
            response = await self.client.post("/executor/set_proxy", json={"proxy": proxy})
            return response.status_code == 200
        except Exception:
            return False

    async def close(self):
        await self.client.aclose()

import re
import json
import asyncio
from typing import Optional, Dict, Any
from .browserwing_client import BrowserWingClient
from .captcha_detector import CaptchaDetector
from .human_behavior import random_delay, human_scroll
from bot.services.captcha_notifier import CaptchaNotifier
from .proxy_rotator import ProxyRotator


class OzonParser:
    PATTERNS = [r"https?://(?:www\.)?ozon\.ru/.*", r"https?://(?:www\.)?ozon\.ru/t/.*"]

    def __init__(self, notifier: Optional[CaptchaNotifier] = None, proxy_rotator: Optional[ProxyRotator] = None):
        self.browser = BrowserWingClient()
        self.detector = CaptchaDetector()
        self.notifier = notifier or CaptchaNotifier()
        self.proxy_rotator = proxy_rotator

    def is_valid(self, url: str) -> bool:
        return any(re.match(p, url) for p in self.PATTERNS)

    async def parse(self, url: str) -> Optional[Dict]:
        for attempt in range(2):
            try:
                ok = await self.browser.navigate(url, wait=7.0)
                if not ok:
                    if attempt == 0 and self.proxy_rotator:
                        await self._rotate_proxy()
                        continue
                    return None

                await human_scroll(self.browser)
                await random_delay(1.0, 2.0)

                captcha = await self.detector.check(self.browser)
                if captcha and captcha.get("detected"):
                    await self.notifier.notify(url, captcha["type"], "ozon")
                    if attempt == 0 and self.proxy_rotator:
                        await self._rotate_proxy()
                        continue
                    return {"error": "captcha", "captcha_type": captcha["type"], "url": url, "source": "ozon"}

                product = await self._extract_jsonld()
                if product:
                    return product
                product = await self._extract_nuxt()
                if product:
                    return product
                product = await self._extract_dom()
                if product:
                    return product

                print("[OzonParser] Не удалось извлечь данные")
                return None
            except Exception as e:
                print(f"[OzonParser] Ошибка: {e}")
                if attempt == 0 and self.proxy_rotator:
                    await self._rotate_proxy()
                    continue
                return None
            finally:
                await self.browser.close()

    async def _rotate_proxy(self):
        proxy = await self.proxy_rotator.get_working_proxy()
        if proxy:
            print(f"[OzonParser] 🔄 Смена прокси: {proxy}")
            await self.browser.close()
            self.browser = BrowserWingClient(proxy=proxy)
            await self.browser.set_browser_proxy(proxy)

    async def _extract_jsonld(self) -> Optional[Dict]:
        script = """
            const scripts = document.querySelectorAll('script[type="application/ld+json"]');
            for (const s of scripts) {
                try {
                    const data = JSON.parse(s.textContent);
                    if (data['@type'] === 'Product') return JSON.stringify(data);
                    if (Array.isArray(data)) {
                        const p = data.find(x => x['@type'] === 'Product');
                        if (p) return JSON.stringify(p);
                    }
                } catch(e) {}
            }
            return null;
        """
        result = await self.browser.execute_js(script)
        if not result:
            return None
        try:
            data = json.loads(result) if isinstance(result, str) else result
            offers = data.get("offers", {})
            if isinstance(offers, list) and offers:
                offers = offers[0]
            elif not isinstance(offers, dict):
                offers = {}
            img = data.get("image", "")
            if isinstance(img, list) and img:
                img = img[0]
            return {
                "name": data.get("name", "Неизвестно"),
                "price": self._clean_price(offers.get("price")),
                "old_price": self._clean_price(offers.get("highPrice")),
                "image": img or "",
                "url": data.get("url", ""),
                "source": "ozon",
            }
        except Exception as e:
            print(f"[OzonParser] JSON-LD ошибка: {e}")
            return None

    async def _extract_nuxt(self) -> Optional[Dict]:
        script = """
            if (window.__NUXT__ && window.__NUXT__.state) {
                return JSON.stringify(window.__NUXT__.state);
            }
            return null;
        """
        result = await self.browser.execute_js(script)
        if not result:
            return None
        try:
            state = json.loads(result) if isinstance(result, str) else result
            product = self._find_in_state(state)
            if not product:
                return None
            return {
                "name": product.get("name", product.get("title", "Неизвестно")),
                "price": self._clean_price(product.get("price") or product.get("finalPrice")),
                "old_price": self._clean_price(product.get("oldPrice") or product.get("originalPrice")),
                "image": product.get("image", product.get("coverImage", "")),
                "url": product.get("url", ""),
                "source": "ozon",
            }
        except Exception as e:
            print(f"[OzonParser] NUXT ошибка: {e}")
            return None

    def _find_in_state(self, obj: Any) -> Optional[Dict]:
        if not isinstance(obj, dict):
            return None
        if "name" in obj and ("price" in obj or "finalPrice" in obj):
            return obj
        for v in obj.values():
            if isinstance(v, dict):
                found = self._find_in_state(v)
                if found:
                    return found
            elif isinstance(v, list):
                for item in v:
                    found = self._find_in_state(item)
                    if found:
                        return found
        return None

    async def _extract_dom(self) -> Optional[Dict]:
        script = """
            const nameEl = document.querySelector('h1') || document.querySelector('[data-widget="webProductHeading"] h1');
            const priceEl = document.querySelector('span[data-testid="price"]') || document.querySelector('[data-widget="webPrice"] span');
            const oldPriceEl = document.querySelector('span[data-testid="old-price"]');
            const imgEl = document.querySelector('img[data-testid="image"]');
            return JSON.stringify({
                name: nameEl ? nameEl.innerText.trim() : null,
                price: priceEl ? priceEl.innerText.trim() : null,
                old_price: oldPriceEl ? oldPriceEl.innerText.trim() : null,
                image: imgEl ? imgEl.src : null
            });
        """
        result = await self.browser.execute_js(script)
        if not result:
            return None
        try:
            data = json.loads(result) if isinstance(result, str) else result
            return {
                "name": data.get("name") or "Неизвестно",
                "price": self._clean_price(data.get("price")),
                "old_price": self._clean_price(data.get("old_price")),
                "image": data.get("image") or "",
                "url": "",
                "source": "ozon",
            }
        except Exception:
            return None

    def _clean_price(self, raw) -> Optional[int]:
        if raw is None:
            return None
        if isinstance(raw, (int, float)):
            return int(raw)
        digits = re.sub(r"[^\d]", "", str(raw))
        return int(digits) if digits else None

    async def close(self):
        await self.browser.close()

import re
import asyncio
from typing import Optional, Dict, Any
from .browserwing_client import BrowserWingClient
from .captcha_detector import CaptchaDetector
from .human_behavior import random_delay, human_scroll, human_mouse_move
from bot.services.captcha_notifier import CaptchaNotifier
from .proxy_rotator import ProxyRotator


class UniversalBrowserParser:
    def __init__(self, notifier: Optional[CaptchaNotifier] = None, proxy_rotator: Optional[ProxyRotator] = None):
        self.browser = BrowserWingClient()
        self.detector = CaptchaDetector()
        self.notifier = notifier or CaptchaNotifier()
        self.proxy_rotator = proxy_rotator

    async def parse(self, url: str, selectors: Dict[str, str], max_retries: int = 2) -> Optional[Dict]:
        for attempt in range(max_retries):
            try:
                print(f"[UniversalParser] Попытка {attempt + 1}/{max_retries}: {url}")
                ok = await self.browser.navigate(url, wait=5.0)
                if not ok:
                    if attempt < max_retries - 1 and self.proxy_rotator:
                        await self._rotate_proxy()
                        continue
                    return {"error": "navigate_failed", "url": url}

                await human_scroll(self.browser)
                await human_mouse_move(self.browser)
                await random_delay(1.0, 3.0)

                captcha = await self.detector.check(self.browser)
                if captcha and captcha.get("detected"):
                    await self.notifier.notify(url, captcha["type"], "universal")
                    if attempt < max_retries - 1 and self.proxy_rotator:
                        await self._rotate_proxy()
                        continue
                    return {"error": "captcha", "captcha_type": captcha["type"], "url": url, "source": "universal"}

                result = {"url": url, "source": "universal"}
                for field, selector in selectors.items():
                    script = f"const el = document.querySelector('{selector}'); return el ? el.innerText.trim() : null;"
                    value = await self.browser.execute_js(script)
                    result[field] = value if value else "Не найдено"

                for price_field in ["price", "old_price"]:
                    if price_field in result and result[price_field] and result[price_field] != "Не найдено":
                        digits = re.sub(r"[^\d]", "", str(result[price_field]))
                        result[price_field] = int(digits) if digits else None
                    else:
                        result[price_field] = None

                return result

            except Exception as e:
                print(f"[UniversalParser] Ошибка: {e}")
                if attempt < max_retries - 1 and self.proxy_rotator:
                    await self._rotate_proxy()
                    continue
                return {"error": str(e), "url": url}
            finally:
                await self.browser.close()

        return {"error": "max_retries_exceeded", "url": url}

    async def _rotate_proxy(self):
        proxy = await self.proxy_rotator.get_working_proxy()
        if proxy:
            print(f"[UniversalParser] 🔄 Смена прокси: {proxy}")
            await self.browser.close()
            self.browser = BrowserWingClient(proxy=proxy)
            await self.browser.set_browser_proxy(proxy)

    async def parse_with_js(self, url: str, js_script: str) -> Optional[Dict]:
        try:
            ok = await self.browser.navigate(url, wait=5.0)
            if not ok:
                return None
            result = await self.browser.execute_js(js_script)
            if isinstance(result, str):
                import json
                return json.loads(result)
            return result
        except Exception as e:
            print(f"[UniversalParser] Ошибка JS: {e}")
            return None
        finally:
            await self.browser.close()
