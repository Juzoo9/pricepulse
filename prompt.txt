import httpx
import asyncio
import json
from typing import Optional, Dict, Any


class BrowserWingClient:
    """Клиент для BrowserWing Executor API."""

    def __init__(self, base_url: str = "http://127.0.0.1:8080/api/v1"):
        self.client = httpx.AsyncClient(base_url=base_url, timeout=60.0)

    async def navigate(self, url: str, wait: float = 5.0) -> bool:
        try:
            response = await self.client.post("/executor/navigate", json={"url": url})
            response.raise_for_status()
            await asyncio.sleep(wait)
            return True
        except Exception as e:
            print(f"[BrowserWing] Ошибка navigate: {e}")
            return False

    async def execute_js(self, script: str) -> Any:
        try:
            response = await self.client.post("/executor/execute_script", json={"script": script})
            response.raise_for_status()
            data = response.json()
            return data
        except Exception as e:
            print(f"[BrowserWing] Ошибка execute_js: {e}")
            return None

    async def snapshot(self) -> Dict[str, Any]:
        try:
            response = await self.client.get("/executor/snapshot")
            response.raise_for_status()
            return response.json()
        except Exception as e:
            print(f"[BrowserWing] Ошибка snapshot: {e}")
            return {}

    async def close(self):
        await self.client.aclose()

import re
import asyncio
from typing import Optional, Dict, Any
from .browserwing_client import BrowserWingClient


class UniversalBrowserParser:
    """Универсальный парсер для любого сайта через BrowserWing."""

    def __init__(self):
        self.browser = BrowserWingClient()

    async def parse(self, url: str, selectors: Dict[str, str]) -> Optional[Dict]:
        try:
            ok = await self.browser.navigate(url, wait=5.0)
            if not ok:
                return None

            result = {"url": url, "source": "universal"}
            for field, selector in selectors.items():
                script = f"""
                    const el = document.querySelector('{selector}');
                    return el ? el.innerText.trim() : null;
                """
                value = await self.browser.execute_js(script)
                result[field] = value if value else "Не найдено"

            if "price" in result and result["price"] and result["price"] != "Не найдено":
                digits = re.sub(r"[^\d]", "", str(result["price"]))
                result["price"] = int(digits) if digits else None
            else:
                result["price"] = None

            if "old_price" in result and result["old_price"] and result["old_price"] != "Не найдено":
                digits = re.sub(r"[^\d]", "", str(result["old_price"]))
                result["old_price"] = int(digits) if digits else None
            else:
                result["old_price"] = None

            return result
        except Exception as e:
            print(f"[UniversalParser] Ошибка: {e}")
            return None
        finally:
            await self.browser.close()

    async def parse_with_js(self, url: str, js_script: str) -> Optional[Dict]:
        try:
            ok = await self.browser.navigate(url, wait=5.0)
            if not ok:
                return None
            result = await self.browser.execute_js(js_script)
            if isinstance(result, str):
                import json
                return json.loads(result)
            return result
        except Exception as e:
            print(f"[UniversalParser] Ошибка JS: {e}")
            return None
        finally:
            await self.browser.close()

import re
import json
import asyncio
from typing import Optional, Dict, Any
from .browserwing_client import BrowserWingClient


class OzonParser:
    PATTERNS = [
        r"https?://(?:www\.)?ozon\.ru/.*",
        r"https?://(?:www\.)?ozon\.ru/t/.*",
    ]

    def __init__(self):
        self.browser = BrowserWingClient()

    def is_valid(self, url: str) -> bool:
        return any(re.match(p, url) for p in self.PATTERNS)

    async def parse(self, url: str) -> Optional[Dict]:
        try:
            ok = await self.browser.navigate(url, wait=7.0)
            if not ok:
                return None

            product = await self._extract_jsonld()
            if product:
                return product

            product = await self._extract_nuxt()
            if product:
                return product

            product = await self._extract_dom()
            if product:
                return product

            print("[OzonParser] Не удалось извлечь данные")
            return None
        except Exception as e:
            print(f"[OzonParser] Ошибка: {e}")
            return None
        finally:
            await self.browser.close()

    async def _extract_jsonld(self) -> Optional[Dict]:
        script = """
            const scripts = document.querySelectorAll('script[type="application/ld+json"]');
            for (const s of scripts) {
                try {
                    const data = JSON.parse(s.textContent);
                    if (data['@type'] === 'Product') return JSON.stringify(data);
                    if (Array.isArray(data)) {
                        const p = data.find(x => x['@type'] === 'Product');
                        if (p) return JSON.stringify(p);
                    }
                } catch(e) {}
            }
            return null;
        """
        result = await self.browser.execute_js(script)
        if not result:
            return None
        try:
            data = json.loads(result) if isinstance(result, str) else result
            offers = data.get("offers", {})
            if isinstance(offers, list) and offers:
                offers = offers[0]
            elif not isinstance(offers, dict):
                offers = {}
            img = data.get("image", "")
            if isinstance(img, list) and img:
                img = img[0]
            return {
                "name": data.get("name", "Неизвестно"),
                "price": self._clean_price(offers.get("price")),
                "old_price": self._clean_price(offers.get("highPrice")),
                "image": img or "",
                "url": data.get("url", ""),
                "source": "ozon",
            }
        except Exception as e:
            print(f"[OzonParser] JSON-LD ошибка: {e}")
            return None

    async def _extract_nuxt(self) -> Optional[Dict]:
        script = """
            if (window.__NUXT__ && window.__NUXT__.state) {
                return JSON.stringify(window.__NUXT__.state);
            }
            return null;
        """
        result = await self.browser.execute_js(script)
        if not result:
            return None
        try:
            state = json.loads(result) if isinstance(result, str) else result
            product = self._find_in_state(state)
            if not product:
                return None
            return {
                "name": product.get("name", product.get("title", "Неизвестно")),
                "price": self._clean_price(product.get("price") or product.get("finalPrice")),
                "old_price": self._clean_price(product.get("oldPrice") or product.get("originalPrice")),
                "image": product.get("image", product.get("coverImage", "")),
                "url": product.get("url", ""),
                "source": "ozon",
            }
        except Exception as e:
            print(f"[OzonParser] NUXT ошибка: {e}")
            return None

    def _find_in_state(self, obj: Any) -> Optional[Dict]:
        if not isinstance(obj, dict):
            return None
        if "name" in obj and ("price" in obj or "finalPrice" in obj):
            return obj
        for v in obj.values():
            if isinstance(v, dict):
                found = self._find_in_state(v)
                if found:
                    return found
            elif isinstance(v, list):
                for item in v:
                    found = self._find_in_state(item)
                    if found:
                        return found
        return None

    async def _extract_dom(self) -> Optional[Dict]:
        script = """
            const nameEl = document.querySelector('h1') || document.querySelector('[data-widget="webProductHeading"] h1');
            const priceEl = document.querySelector('span[data-testid="price"]') || document.querySelector('[data-widget="webPrice"] span');
            const oldPriceEl = document.querySelector('span[data-testid="old-price"]');
            const imgEl = document.querySelector('img[data-testid="image"]');
            return JSON.stringify({
                name: nameEl ? nameEl.innerText.trim() : null,
                price: priceEl ? priceEl.innerText.trim() : null,
                old_price: oldPriceEl ? oldPriceEl.innerText.trim() : null,
                image: imgEl ? imgEl.src : null
            });
        """
        result = await self.browser.execute_js(script)
        if not result:
            return None
        try:
            data = json.loads(result) if isinstance(result, str) else result
            return {
                "name": data.get("name") or "Неизвестно",
                "price": self._clean_price(data.get("price")),
                "old_price": self._clean_price(data.get("old_price")),
                "image": data.get("image") or "",
                "url": "",
                "source": "ozon",
            }
        except Exception:
            return None

    def _clean_price(self, raw) -> Optional[int]:
        if raw is None:
            return None
        if isinstance(raw, (int, float)):
            return int(raw)
        digits = re.sub(r"[^\d]", "", str(raw))
        return int(digits) if digits else None

    async def close(self):
        await self.browser.close()

import asyncio
from typing import Optional, Dict

class WildberriesParser:
    PATTERNS = [
        r"https?://(?:www\.)?wildberries\.ru/catalog/\d+/detail\.aspx.*",
        r"https?://(?:www\.)?wildberries\.ru/.*",
    ]

    def __init__(self):
        self.browser = BrowserWingClient()

    def is_valid(self, url: str) -> bool:
        return any(re.match(p, url) for p in self.PATTERNS)

    async def parse(self, url: str) -> Optional[Dict]:
        try:
            ok = await self.browser.navigate(url, wait=6.0)
            if not ok:
                return None

            script = """
                try {
                    const p = window.__INITIAL_STATE__?.catalog?.product;
                    if (p) return JSON.stringify(p);
                    return null;
                } catch(e) { return null; }
            """
            result = await self.browser.execute_js(script)
            if not result:
                return await self._extract_from_dom()

            data = json.loads(result) if isinstance(result, str) else result
            price_raw = data.get("salePriceU") or data.get("price") or data.get("priceU")
            price = int(price_raw) / 100 if price_raw else None

            old_raw = data.get("priceU") or data.get("oldPrice")
            old_price = int(old_raw) / 100 if old_raw else None

            return {
                "name": data.get("name", data.get("goodsName", "Неизвестно")),
                "price": price,
                "old_price": old_price,
                "image": data.get("image", ""),
                "url": url,
                "source": "wildberries",

        except Exception as e:
            print(f"[WildberriesParser] Ошибка: {e}")
            return None
        finally:

            data = json.loads(result) if isinstance(result, str) else result

                "name": data.get("name") or "Неизвестно",
                "price": self._clean_price(data.get("price")),

                "image": data.get("image") or "",

            top=Side(style='thin'), bottom=Side(style='thin')
        )

        for col_num, header in enumerate(headers, 1):
            cell = ws.cell(row=1, column=col_num, value=header)
            cell.fill = header_fill
            cell.font = header_font
            cell.alignment = Alignment(horizontal="center", vertical="center")
            cell.border = thin_border

        # Данные
        for row_num, item in enumerate(data, 2):
            row_data = [
                item.get("name", ""),
                item.get("price", ""),
                item.get("old_price", "") or "",
                item.get("image", ""),
                item.get("url", ""),
                item.get("source", ""),
            ]
            for col_num, value in enumerate(row_data, 1):
                cell = ws.cell(row=row_num, column=col_num, value=value)
                cell.border = thin_border
                cell.alignment = Alignment(vertical="center", wrap_text=True)
                if col_num == 2 and value:
                    cell.number_format = '#,##0 "₽"'

        # Автоширина колонок
        for col in range(1, len(headers) + 1):
            max_length = 0
            column = get_column_letter(col)
            for row in ws.iter_rows(min_col=col, max_col=col):
                for cell in row:
                    try:
                        if cell.value:
                            max_length = max(max_length, len(str(cell.value)))
                    except:
                        pass
            adjusted_width = min(max_length + 2, 60)
            ws.column_dimensions[column].width = adjusted_width

        # Фиксация заголовка
        ws.freeze_panes = "A2"

        # Автофильтр
        ws.auto_filter.ref = ws.dimensions

        wb.save(self.output_path)
        return os.path.abspath(self.output_path)

import os
from typing import List, Dict, Optional
import gspread
from gspread.utils import rowcol_to_a1
from google.oauth2.service_account import Credentials


class GoogleSheetsExporter:
    """Экспорт результатов парсинга в Google Sheets."""

    SCOPES = [
        "https://www.googleapis.com/auth/spreadsheets",
        "https://www.googleapis.com/auth/drive",
    ]

    def __init__(self, credentials_path: str = "google_credentials.json"):
        self.credentials_path = credentials_path
        self.client = None

    def _get_client(self) -> gspread.Client:
        if self.client is None:
            if not os.path.exists(self.credentials_path):
                raise FileNotFoundError(
                    f"Файл credentials не найден: {self.credentials_path}. "
                    "Создай Service Account в Google Cloud Console, скачай JSON-ключ и положи в проект."
                )
            creds = Credentials.from_service_account_file(self.credentials_path, scopes=self.SCOPES)
            self.client = gspread.authorize(creds)
        return self.client

    def export(self, data: List[Dict], spreadsheet_name: str = "PricePulse Results") -> str:
        """
        Создаёт Google-таблицу, заполняет данными и возвращает URL.
        """
        client = self._get_client()

        # Создаём таблицу
        spreadsheet = client.create(spreadsheet_name)
        worksheet = spreadsheet.sheet1
        worksheet.title = "Парсинг"

        # Заголовки
        headers = ["Название", "Цена", "Старая цена", "Изображение", "URL", "Источник"]
        worksheet.append_row(headers)

        # Форматирование заголовков
        worksheet.format("A1:F1", {
            "textFormat": {"bold": True, "fontSize": 12},
            "backgroundColor": {"red": 0.21, "green": 0.38, "blue": 0.57},
            "horizontalAlignment": "CENTER",
        })

        # Данные
        for item in data:
            row = [
                item.get("name", ""),
                item.get("price", "") if item.get("price") is not None else "",
                item.get("old_price", "") if item.get("old_price") is not None else "",
                item.get("image", ""),
                item.get("url", ""),
                item.get("source", ""),
            ]
            worksheet.append_row(row)

        # Автоширина (примерная)
        worksheet.columns_auto_resize(0, 5)

        # Публичный доступ (опционально — можно убрать)
        spreadsheet.share("", perm_type="anyone", role="reader")

        return spreadsheet.url
